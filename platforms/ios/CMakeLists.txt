cmake_minimum_required(VERSION 3.28)

project(ARMSX3iOS LANGUAGES C CXX OBJC OBJCXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    message(FATAL_ERROR "ARMSX3iOS must be configured with -DCMAKE_SYSTEM_NAME=iOS")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)

set(ARMSX3_CORE_LIBRARY "" CACHE FILEPATH "Path to libRPCS3Core.dylib")
if(NOT EXISTS "${ARMSX3_CORE_LIBRARY}")
    message(FATAL_ERROR "Set ARMSX3_CORE_LIBRARY to a built libRPCS3Core.dylib")
endif()

get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

set(ARMSX3_IOS_CONTROLLER_ARTWORK
    app/ControllerSkins/controller-ps3-landscape-left.png
    app/ControllerSkins/controller-ps3-landscape-right.png
)
set_source_files_properties(${ARMSX3_IOS_CONTROLLER_ARTWORK} PROPERTIES
    MACOSX_PACKAGE_LOCATION Resources
)

add_library(RPCS3Core SHARED IMPORTED GLOBAL)
set_target_properties(RPCS3Core PROPERTIES IMPORTED_LOCATION "${ARMSX3_CORE_LIBRARY}")

add_executable(ARMSX3iOS MACOSX_BUNDLE
    app/main.mm
    app/ARMSX3AppDelegate.mm
    app/ARMSX3CoreSession.mm
    app/ARMSX3MetalView.mm
    app/ARMSX3ViewController.mm
    ${ARMSX3_IOS_CONTROLLER_ARTWORK}
)

target_include_directories(ARMSX3iOS PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/app"
    "${REPO_ROOT}/rpcs3/ios"
)

target_link_libraries(ARMSX3iOS PRIVATE
    RPCS3Core
    "-framework CoreGraphics"
    "-framework Foundation"
    "-framework GameController"
    "-framework Metal"
    "-framework QuartzCore"
    "-framework UIKit"
    "-framework UniformTypeIdentifiers"
)

add_custom_command(TARGET ARMSX3iOS POST_BUILD
    COMMAND /bin/sh "${CMAKE_CURRENT_SOURCE_DIR}/app/embed-rpcs3-core.sh"
        "${ARMSX3_CORE_LIBRARY}"
    COMMENT "Embedding the current RPCS3Core build"
    VERBATIM
)
set_property(TARGET ARMSX3iOS APPEND PROPERTY LINK_DEPENDS "${ARMSX3_CORE_LIBRARY}")

set_target_properties(ARMSX3iOS PROPERTIES
    MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/app/Info.plist.in"
    XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC YES
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO
    XCODE_ATTRIBUTE_CODE_SIGNING_REQUIRED NO
    XCODE_ATTRIBUTE_COMPILER_INDEX_STORE_ENABLE NO
    XCODE_ATTRIBUTE_CURRENT_PROJECT_VERSION 12
    XCODE_ATTRIBUTE_DEBUG_INFORMATION_FORMAT dwarf
    XCODE_ATTRIBUTE_ENABLE_BITCODE NO
    XCODE_ATTRIBUTE_GENERATE_INFOPLIST_FILE NO
    XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET "${CMAKE_OSX_DEPLOYMENT_TARGET}"
    XCODE_ATTRIBUTE_LD_RUNPATH_SEARCH_PATHS "@executable_path/Frameworks"
    XCODE_ATTRIBUTE_MARKETING_VERSION "0.13.0"
    XCODE_ATTRIBUTE_ONLY_ACTIVE_ARCH YES
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "com.thec0de.armsx3ios"
    XCODE_ATTRIBUTE_SUPPORTED_PLATFORMS iphoneos
    XCODE_ATTRIBUTE_SUPPORTS_MACCATALYST NO
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY 1
)
